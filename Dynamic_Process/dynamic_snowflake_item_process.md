 CREATE OR REPLACE PROCEDURE DIM_DYNAMIC_PROCESS_SP(
    P_TABLE_NAME VARCHAR,
    P_OPERATION VARCHAR
)
RETURNS VARCHAR
LANGUAGE SQL
AS
DECLARE
    V_SOURCE_TABLE       VARCHAR;
    V_TARGET_TABLE       VARCHAR;
    V_COLUMN_LIST        VARCHAR;
    V_UPDATE_LIST        VARCHAR;
    V_PK_COLUMN          VARCHAR;
    V_SQL                 VARCHAR;
    V_START_TIME         TIMESTAMP;
    V_END_TIME           TIMESTAMP;
    V_SOURCE_COUNT       NUMBER;
    V_TARGET_COUNT       NUMBER;
BEGIN

    /* Get source and target table from control table */
    SELECT SOURCE_TABLE_NAME, TARGET_TABLE_NAME
    INTO :V_SOURCE_TABLE, :V_TARGET_TABLE
    FROM QUERY_EXECUTION_CONTROL_TABLE
    WHERE DATA_SUBJECT = :P_TABLE_NAME
      AND OPERATION = UPPER(:P_OPERATION)
    LIMIT 1;

    V_START_TIME := CURRENT_TIMESTAMP();

    /* Get primary key dynamically */
    DESC TABLE IDENTIFIER(:V_SOURCE_TABLE);

    SELECT "name"
    INTO :V_PK_COLUMN
    FROM TABLE(RESULT_SCAN(LAST_QUERY_ID()))
    WHERE "primary key" = 'Y'
    LIMIT 1;

    /* Get all column names dynamically */
    SELECT LISTAGG('"' || COLUMN_NAME || '"', ', ')
           WITHIN GROUP (ORDER BY ORDINAL_POSITION)
    INTO :V_COLUMN_LIST
    FROM INFORMATION_SCHEMA.COLUMNS
    WHERE TABLE_NAME = SPLIT_PART(:V_SOURCE_TABLE, '.', 3)
      AND TABLE_SCHEMA = SPLIT_PART(:V_SOURCE_TABLE, '.', 2)
      AND TABLE_CATALOG = SPLIT_PART(:V_SOURCE_TABLE, '.', 1);

    /* Build UPDATE SET clause dynamically */
    SELECT LISTAGG(
        'T."' || COLUMN_NAME || '" = S."' || COLUMN_NAME || '"',
        ', '
    ) WITHIN GROUP (ORDER BY ORDINAL_POSITION)
    INTO :V_UPDATE_LIST
    FROM INFORMATION_SCHEMA.COLUMNS
    WHERE TABLE_NAME = SPLIT_PART(:V_SOURCE_TABLE, '.', 3)
      AND TABLE_SCHEMA = SPLIT_PART(:V_SOURCE_TABLE, '.', 2)
      AND TABLE_CATALOG = SPLIT_PART(:V_SOURCE_TABLE, '.', 1)
      AND COLUMN_NAME <> :V_PK_COLUMN;

    /* INSERT */
    IF (UPPER(:P_OPERATION) = 'INSERT') THEN

        V_SQL :=
            'INSERT INTO ' || V_TARGET_TABLE ||
            ' (' || V_COLUMN_LIST || ')' ||
            ' SELECT ' || V_COLUMN_LIST ||
            ' FROM ' || V_SOURCE_TABLE || ' S' ||
            ' WHERE NOT EXISTS (' ||
            ' SELECT 1 FROM ' || V_TARGET_TABLE || ' T' ||
            ' WHERE T."' || V_PK_COLUMN || '" = S."' || V_PK_COLUMN || '"' ||
            ' )';

        EXECUTE IMMEDIATE V_SQL;

    /* UPDATE */
    ELSEIF (UPPER(:P_OPERATION) = 'UPDATE') THEN

        V_SQL :=
            'UPDATE ' || V_TARGET_TABLE || ' T SET ' ||
            V_UPDATE_LIST ||
            ' FROM ' || V_SOURCE_TABLE || ' S' ||
            ' WHERE T."' || V_PK_COLUMN || '" = S."' || V_PK_COLUMN || '"';

        EXECUTE IMMEDIATE V_SQL;

    /* DELETE */
    ELSEIF (UPPER(:P_OPERATION) = 'DELETE') THEN

        V_SQL :=
            'DELETE FROM ' || V_TARGET_TABLE || ' T ' ||
            ' WHERE NOT EXISTS (' ||
            ' SELECT 1 FROM ' || V_SOURCE_TABLE || ' S' ||
            ' WHERE T."' || V_PK_COLUMN || '" = S."' || V_PK_COLUMN || '"' ||
            ' )';

        EXECUTE IMMEDIATE V_SQL;

    ELSE

        RETURN 'INVALID OPERATION. USE INSERT, UPDATE OR DELETE';

    END IF;

    /* Get record counts */
    SELECT COUNT(*)
    INTO :V_SOURCE_COUNT
    FROM IDENTIFIER(:V_SOURCE_TABLE);

    SELECT COUNT(*)
    INTO :V_TARGET_COUNT
    FROM IDENTIFIER(:V_TARGET_TABLE);

    V_END_TIME := CURRENT_TIMESTAMP();

    /* Audit */
    INSERT INTO EDW_SRC.PUBLIC.EDW_AUDIT_LOG
    (
        SOURCE_TABLE_NAME,
        TARGET_TABLE_NAME,
        SOURCE_RECORD_COUNT,
        TARGET_RECORD_COUNT,
        START_TIME,
        END_TIME,
        JOB_STATUS,
        REASON_FOR_FAILURE
    )
    VALUES
    (
        :V_SOURCE_TABLE,
        :V_TARGET_TABLE,
        :V_SOURCE_COUNT,
        :V_TARGET_COUNT,
        :V_START_TIME,
        :V_END_TIME,
        'SUCCESS',
        NULL
    );

    RETURN :P_TABLE_NAME || ' ' || :P_OPERATION ||
           ' completed successfully';

EXCEPTION
    WHEN OTHER THEN

        V_END_TIME := CURRENT_TIMESTAMP();

        INSERT INTO EDW_AUDIT_LOG
        (
            SOURCE_TABLE_NAME,
            TARGET_TABLE_NAME,
            SOURCE_RECORD_COUNT,
            TARGET_RECORD_COUNT,
            START_TIME,
            END_TIME,
            JOB_STATUS,
            REASON_FOR_FAILURE
        )
        VALUES
        (
            :V_SOURCE_TABLE,
            :V_TARGET_TABLE,
            :V_SOURCE_COUNT,
            :V_TARGET_COUNT,
            :V_START_TIME,
            :V_END_TIME,
            'FAILED',
            :SQLERRM
        );

        RETURN :P_TABLE_NAME || ' ' || :P_OPERATION ||
               ' FAILED: ' || :SQLERRM;
END;

CALL DIM_DYNAMIC_PROCESS_SP('ITEM', 'INSERT');

SELECT *
FROM EDW_SRC.PUBLIC.EDW_AUDIT_LOG
ORDER BY START_TIME DESC;
